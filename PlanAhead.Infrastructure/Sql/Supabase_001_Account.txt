create table public.accounts
(
    id uuid primary key,

    user_id uuid not null references auth.users(id) on delete cascade,

    name text not null,
    description text not null default '',

    opening_balance numeric(18,2) not null default 0,

    display_order integer not null default 0,

    notes text not null default '',

    archived boolean not null default false,

    icon_id text not null default 'Bank',

    created_utc timestamptz not null default now(),

    updated_utc timestamptz not null default now(),

    deleted_utc timestamptz null
);

create index idx_accounts_user
on public.accounts(user_id);

create index idx_accounts_deleted
on public.accounts(deleted_utc);

create index idx_accounts_display_order
on public.accounts(user_id, display_order);

create or replace function public.set_updated_utc()
returns trigger
language plpgsql
as $$
begin
    new.updated_utc = now();
    return new;
end;
$$;

create trigger trg_accounts_updated
before update
on public.accounts
for each row
execute function public.set_updated_utc();

alter table public.accounts
enable row level security;

create policy "Users can view their own accounts"
on public.accounts
for select
using (
    auth.uid() = user_id
);

create policy "Users can insert accounts"
on public.accounts
for insert
to authenticated
with check (true);

create policy "Users can update their own accounts"
on public.accounts
for update
using (
    auth.uid() = user_id
)
with check (
    auth.uid() = user_id
);

create policy "Users can delete their own accounts"
on public.accounts
for delete
using (
    auth.uid() = user_id
);

create or replace function public.set_account_user_id()
returns trigger
language plpgsql
as $$
begin
    new.user_id := auth.uid();
    return new;
end;
$$;


create trigger trg_accounts_set_user
before insert
on public.accounts
for each row
execute function public.set_account_user_id();
