create table public.funds
(
    id uuid primary key,

    user_id uuid not null references auth.users(id) on delete cascade,

    account_id uuid not null references public.accounts(id) on delete cascade,

    name text not null,

    description text not null default '',

    frequency integer not null,

    status integer not null,

    display_order integer not null default 0,

    notes text not null default '',

    icon_id text not null default 'PiggyBank',

    created_utc timestamptz not null default now(),

    updated_utc timestamptz not null default now(),

    deleted_utc timestamptz null
);

create index idx_funds_user
on public.funds(user_id);

create index idx_funds_account
on public.funds(account_id);

create index idx_funds_deleted
on public.funds(deleted_utc);

create index idx_funds_display_order
on public.funds(account_id, display_order);


create trigger trg_funds_updated
before update
on public.funds
for each row
execute function public.set_updated_utc();


alter table public.funds
enable row level security;


create policy "Users can view their own funds"
on public.funds
for select
using (
    auth.uid() = user_id
);

create policy "Users can insert funds"
on public.funds
for insert
to authenticated
with check (true);

create policy "Users can update their own funds"
on public.funds
for update
using (
    auth.uid() = user_id
)
with check (
    auth.uid() = user_id
);


create policy "Users can delete their own funds"
on public.funds
for delete
using (
    auth.uid() = user_id
);



create or replace function public.set_fund_user_id()
returns trigger
language plpgsql
as $$
declare
    account_owner uuid;
begin

    select user_id
      into account_owner
      from public.accounts
     where id = new.account_id
       and deleted_utc is null;

    if account_owner is null then
        raise exception 'Account does not exist.';
    end if;

    if account_owner <> auth.uid() then
        raise exception 'You do not own this account.';
    end if;

    new.user_id := account_owner;

    return new;
end;
$$;


create trigger trg_funds_set_user
before insert or update
on public.funds
for each row
execute function public.set_fund_user_id();
